cmake_minimum_required(VERSION 3.16)
project(traffic_flow_simulator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_GUI "Build the raylib dashboard" ON)
if(BUILD_GUI)
  find_package(raylib 5.5 REQUIRED)
  add_executable(traffic_simulator Graphics.cpp)
  target_link_libraries(traffic_simulator PRIVATE raylib)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(traffic_checks tests/core_checks.cpp)
  target_include_directories(traffic_checks PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME traffic_core COMMAND traffic_checks)
endif()
